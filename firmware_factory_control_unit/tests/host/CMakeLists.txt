# Host unit tests of the Factory Control Unit's application code (src/), with GCC 16 on the PC.
#
# rtoos is pulled in as a subproject for its host libraries (rtos_kernel, rtos_filex); hardware is
# replaced by simulations, e.g. sim/sd_card_sim.cpp: the SD card as a FileX RAM disk instead of
# src/sd_card.cpp + SDMMC. Build only through the presets (CMakePresets.json in this directory):
#   cmake --preset host && cmake --build --preset host && ctest --preset host
cmake_minimum_required(VERSION 3.28)
project(fcu_host_tests LANGUAGES C CXX)

set(FCU_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../..)

# as the firmware (and rtoos, which sets it only in its own scope): C++26 with contracts
set(CMAKE_CXX_STANDARD 26)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(RTOS_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${FCU_DIR}/rtoos rtoos)

# ws_protocol.hh is generated by the builder (gitignored, see docs/build-process.md)
set(FCU_GENERATED ${FCU_DIR}/generated)
if(NOT EXISTS ${FCU_GENERATED}/ws_protocol.hh)
    message(FATAL_ERROR "${FCU_GENERATED}/ws_protocol.hh fehlt -- vorher im Projektordner "
                        "'dotnet run --project builder -- GenerateBestBinaryBufferFiles' ausfuehren.")
endif()

include(FetchContent)
FetchContent_Declare(
    googletest
    URL https://github.com/google/googletest/archive/refs/tags/v1.17.0.tar.gz
    DOWNLOAD_EXTRACT_TIMESTAMP ON)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)

add_executable(fcu_host_tests
    missions_test.cpp
    ethercat_sii_test.cpp  # src/ethercat_sii.hpp, header only
    ethercat_registers_test.cpp  # src/ethercat_registers.hpp, header only
    ws2812_test.cpp  # src/ws2812.hpp, header only
    sim/sd_card_sim.cpp
    ${FCU_DIR}/src/roarm/missions.cpp)
# stubs/ first: replaces board/fcu.hpp and system.hpp (hardware, kernel instance) for the code under test
target_include_directories(fcu_host_tests PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}/stubs
    ${CMAKE_CURRENT_SOURCE_DIR}/sim
    ${FCU_DIR}/src
    ${FCU_GENERATED})
target_link_libraries(fcu_host_tests PRIVATE rtos_kernel rtos_filex GTest::gtest_main)

enable_testing()
include(GoogleTest)
gtest_discover_tests(fcu_host_tests DISCOVERY_MODE PRE_TEST)
